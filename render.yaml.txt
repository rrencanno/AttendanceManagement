services:
  - type: web
    name: AttendanceManagement
    runtime: php
    plan: free
    buildCommand: "composer install --optimize-autoloader --no-dev"
    startCommand: "vendor/bin/heroku-php-apache2 public/"
    phpVersion: "8.1" # ★★★ PHPのバージョンを明示的に指定 ★★★
    envVars:
      - key: APP_KEY
        generateValue: true
      - key: APP_ENV
        value: production
      - key: APP_DEBUG
        value: false
      - key: LOG_CHANNEL
        value: stderr
      - key: DB_CONNECTION
        # DB_CONNECTION は fromDatabase で設定される DATABASE_URL を使うので、
        # ここでは DBの種類を指定する (Renderのデフォルトはpgsql)
        value: pgsql
        # もしMySQLアドオンを使う場合は 'mysql'
      - fromDatabase:
          name: attendance-db
          property: connectionString

databases:
  - name: attendance-db
    databaseName: attendance_db
    plan: free
    # (オプション) MySQLを使いたい場合
    # engine: mysql
    # mySQLVersion: 8
